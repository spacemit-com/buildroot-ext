# Common parameter
earlycon=smh
console=ttyAMA0,115200n8
init=/init.tmp
bootdelay=3
baudrate=115200

# Partition table for various media
parts_nand=10480000.nand:1m(uboot),128k(env),128k(envbak),256k(logo),128k(dtb),12m(kernel),64m(rootfs),-(user)
parts_nor=nor0:128k(spl),512k(uboot),64k(env),64k(envbak),128k(logo),128k(dtb),6m(kernel),6m(rootfs),-(user)

## eMMC and SD Card partition table, will be used when generating image.
parts_mmc=512k(spl),1m(uboot),512k(env),512k(logo),512k(dtb),16m(kernel),64m(rootfs),-(user)

# partition table for burn firmware with TF card
burn_mmc=512k(spl),2m(uboot),512k(env),512k(logo),-(user)

# Nand flash rootfs device
nand_root=ubi0:rootfs
nand_rootfstype=ubifs

# Nor flash rootfs device
nor_root=/dev/mtdblock6
nor_rootfstype=squashfs

# eMMC/SDCard rootfs device
mmc_rootfstype=ext4

## Get "rootfs" partition number in decimal, and set var "mmc_root"
## Variable "boot_devnum" is set during board_lat_init()
set_mmc_root=part number mmc ${boot_devnum} rootfs rootfs_part; \
	     setexpr rootfs_part ${rootfs_part} + 0; \
	     setenv mmc_root "/dev/mmcblk${boot_devnum}p${rootfs_part}";

# NFS device
# ethaddr=00:00:23:34:45:66
# ipaddr=192.168.1.15
# netmask=255.255.255.0
# serverip=192.168.1.2
# gatewayip=192.168.1.1

preboot=
knl_addr=0x80008000
dtb_addr=0x83000000
mdio_intf=
phyaddr0=1
phy_link_time=10000
netdev=eth0
nfsopts=
nfs_root=
nfs_bootfile=

# Common boot args
commonargs=setenv bootargs earlycon=${earlycon} earlyprintk console=${console} rdinit=${init}

# Nand boot
nand_mtdargs=setenv mtdparts ${parts_nand}

nand_args=run nand_mtdargs; \
	  run commonargs; \
	  setenv bootargs ${bootargs} mtdparts=${parts_nand} ubi.mtd=6 root=${nand_root} rootfstype=${nand_rootfstype}

nand_boot=echo "Try to boot from nand flash..."; \
	  run nand_args; \
	  ubi part kernel && ubi read ${knl_addr} kernel && mtd read dtb ${dtb_addr}; \
	  bootm ${knl_addr} - ${dtb_addr};

# Nor boot
nor_mtdargs=setenv mtdparts ${parts_nor}

nor_args=run nor_mtdargs; \
	 run commonargs; \
	 setenv bootargs ${bootargs} mtdparts=${parts_nor} root=${flashroot} rootfstype=${flashrootfstype}

nor_boot=echo "Try to boot from nor flash..."; \
	 sf probe; \
	 run nor_args; \
	 mtd read kernel ${knl_addr} && mtd read dtb ${dtb_addr}; \
	 bootm ${knl_addr} - ${dtb_addr};

###############################################################################
# eMMC/SDCard boot
###############################################################################
set_mmc_args=run commonargs; \
	     run set_mmc_root; \
	     setenv bootargs "${bootargs}" root=${mmc_root} rootwait rootfstype=${mmc_rootfstype};

## Get "kernel" partition start lba and partition size, and load kernel image
mmc_knl_env=part start mmc ${boot_devnum} kernel knl_lba; \
	    part size mmc ${boot_devnum} kernel knl_size;

mmc_loadknl=run mmc_knl_env; \
	    mmc read ${knl_addr} ${knl_lba} ${knl_size};

## Get "dtb" partition start lba and partition size, and load dtb image
mmc_dtb_env=part start mmc ${boot_devnum} dtb dtb_lba; \
	    part size mmc ${boot_devnum} dtb dtb_size;

mmc_loaddtb=run mmc_dtb_env; mmc read ${dtb_addr} ${dtb_lba} ${dtb_size};

## Do mmc boot
mmc_boot=echo "Try to boot from MMC..."; \
	 run set_mmc_args; \
	 run mmc_loaddtb; \
	 run mmc_loadknl; \
	 bootm ${knl_addr} - ${dtb_addr};

# U-boot autoboot script: try to boot from device, if failed then try nfs boot
# Variable "boot_device" is set during board_late_init()
autoboot=if test ${boot_device} = nand; then \
		run nand_boot; \
	elif test ${boot_device} = nor; then \
		run nor_boot; \
	elif test ${boot_device} = mmc; then \
		run mmc_boot; \
	fi; \
	run nfs_boot;
bootcmd=run autoboot;
